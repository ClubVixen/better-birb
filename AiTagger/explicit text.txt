# --- Helper: Possibly resize the input image ---
def maybe_resize_image(original_path: str, max_dim=1024) -> str:
    try:
        with Image.open(original_path) as im:
            w, h = im.size
            if max(w, h) <= max_dim:
                return original_path
            ratio = max_dim / float(max(w, h))
            new_w = int(w * ratio)
            new_h = int(h * ratio)
            resized = im.resize((new_w, new_h), Image.LANCZOS)
            new_path = original_path + ".resized.png"
            resized.save(new_path, "PNG")
            return new_path
    except Exception as e:
        print(
            f"Warning: could not open/resize {original_path}, using original. Error: {e}"
        )
        return original_path


def load_image(image_path: str) -> Image.Image:
    return Image.open(image_path)


# --- Main ---
def main():
    if len(sys.argv) < 4:
        print(
            "Usage: python deepseek_title.py <image_path> <comma_separated_tags> <rating>"
        )
        sys.exit(1)

    image_path = sys.argv[1]
    booru_tags_str = sys.argv[2].strip()
    rating = sys.argv[3].strip()  # e.g., "Safe", "Suggestive", or "Explicit"

    # Resize image if needed
    resized_path = maybe_resize_image(image_path, max_dim=1024)

    # Choose the base prompt based on rating.
    if rating.lower() == "safe":
        base_prompt = SAFE_BASE_PROMPT
    else:
        base_prompt = NS_BASE_PROMPT

    prompt_text = base_prompt + booru_tags_str

    # Load the VLChatProcessor from DeepSeek‑VL; it also provides the tokenizer.
    vl_processor = VLChatProcessor.from_pretrained(MODEL_PATH)
    model = (
        MultiModalityCausalLM.from_pretrained(
            MODEL_PATH, torch_dtype=DTYPE, trust_remote_code=True
        )
        .to(DEVICE)
        .eval()
    )

    def load_images_for_conversation(conversation):
        """Load all images from a conversation"""
        images = []
        for turn in conversation:
            if "images" not in turn:
                continue
            for img in turn["images"]:
                if isinstance(img, str):
                    images.append(Image.open(img).convert("RGB"))
                elif hasattr(img, "format"):  # PIL Image
                    images.append(img)
        return images

    conversation = [
        {"role": "User", "content": prompt_text, "images": [load_image(resized_path)]},
        {"role": "Assistant", "content": ""},
    ]

    # Prepare the inputs via the VLChatProcessor.
    pil_images = load_images_for_conversation(conversation)
    prepare_inputs = vl_processor(
        conversations=conversation, images=pil_images, force_batchify=True
    ).to(DEVICE)
    # Obtain the image encoder inputs.
    inputs_embeds = model.prepare_inputs_embeds(**prepare_inputs)

    # Generate the title.
    with torch.no_grad():
        outputs = model.language_model.generate(
            inputs_embeds=inputs_embeds,
            attention_mask=prepare_inputs.attention_mask,
            pad_token_id=vl_processor.tokenizer.eos_token_id,
            bos_token_id=vl_processor.tokenizer.bos_token_id,
            eos_token_id=vl_processor.tokenizer.eos_token_id,
            **DEFAULT_GENERATE_PARAMS,
        )

    # DEBUG: Print raw outputs
    print(f"DEBUG - Raw outputs: {outputs}", file=sys.stderr)
    print(f"DEBUG - Outputs shape: {outputs.shape}", file=sys.stderr)

    # Slice off the prompt tokens.
    prompt_len = prepare_inputs.input_ids.shape[1]
    outputs = outputs[:, prompt_len:]
    
    
    # DEBUG: Print decoded output before processing
    raw_result = vl_processor.tokenizer.decode(outputs[0].cpu().tolist())
    print(f"DEBUG - Raw decoded: {raw_result}", file=sys.stderr)
    
    result = raw_result.strip()
    print(f"DEBUG - After strip: {result}", file=sys.stderr)
    # If rating is not safe, apply random word replacements.
    if rating.lower() != "safe":
        result = random_replace_words(result)

    # Remove any trailing quotes or periods.
        output = {
            "title": result if result else "[NO OUTPUT GENERATED]",
            "status": "success" if result else "empty_response",
            "debug": {
                "raw_output": raw_result,
                "processed_output": result
            }
        }


        print(json.dumps(result))  # Ensure this is the ONLY print output
        sys.stdout.flush()  # Force flush the output buffer
        sys.exit(1)

    if __name__ == "__main__":
        main()
